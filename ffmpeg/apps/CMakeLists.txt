# ffmpeg tree: SMEC-style pre-encoded RTP sender / receiver against the distribution's FFmpeg libraries.
cmake_minimum_required(VERSION 3.16)
project(p5g_ffmpeg_apps CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
find_package(Threads REQUIRED)
set(P5G_INCLUDES ${CMAKE_CURRENT_SOURCE_DIR}/common)
add_compile_options(-Wall -Wextra -Wno-unused-parameter -Wno-deprecated-declarations)
add_executable(video_sender sender/video_sender.cc)
target_include_directories(video_sender PRIVATE ${P5G_INCLUDES})
target_link_libraries(video_sender PRIVATE PkgConfig::AV Threads::Threads)
add_executable(video_receiver receiver/video_receiver.cc)
target_include_directories(video_receiver PRIVATE ${P5G_INCLUDES})
target_link_libraries(video_receiver PRIVATE PkgConfig::AV Threads::Threads)
add_executable(depack_test tests/depack_test.cc)
target_include_directories(depack_test PRIVATE ${P5G_INCLUDES})
target_link_libraries(depack_test PRIVATE PkgConfig::AV Threads::Threads)
